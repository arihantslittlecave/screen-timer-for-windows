# Windows version resource embedded into the .exe.
#
# Without this, Windows has no friendly name for the app and falls back to the
# raw filename, so Task Manager, the Startup apps list, the tray icon settings
# and the file's Properties dialog all read "ScreenTimer.exe" instead of the
# app's actual name.
VSVersionInfo(
    ffi=FixedFileInfo(
        filevers=(1, 0, 3, 0),
        prodvers=(1, 0, 3, 0),
        mask=0x3F,
        flags=0x0,
        OS=0x40004,
        fileType=0x1,
        subtype=0x0,
        date=(0, 0),
    ),
    kids=[
        StringFileInfo(
            [
                StringTable(
                    "040904B0",  # US English, Unicode
                    [
                        StringStruct("CompanyName", "arihantslittlecave"),
                        StringStruct("FileDescription", "Screen Timer for Windows"),
                        StringStruct("FileVersion", "1.0.3.0"),
                        StringStruct("InternalName", "ScreenTimer"),
                        StringStruct("LegalCopyright", "MIT License"),
                        StringStruct("OriginalFilename", "ScreenTimer.exe"),
                        StringStruct("ProductName", "Screen Timer for Windows"),
                        StringStruct("ProductVersion", "1.0.3.0"),
                    ],
                )
            ]
        ),
        VarFileInfo([VarStruct("Translation", [0x0409, 1200])]),
    ],
)
